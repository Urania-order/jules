    File: smos/services/experiment_run_service.py

    class ExperimentRunService:
        def __init__(self, db: Session)

        # LIFECYCLE OPERATIONS
        def propose(
            self,
            data: Dict[str, Any],
            created_by: Optional[str] = None,
        ) -> Optional[ExperimentRun]:
            """Create PROPOSED run."""

        def approve(
            self,
            run_id: int,
            approved_by: str,
            reason: Optional[str] = None,
        ) -> Optional[ExperimentRun]:
            """PROPOSED -> APPROVED."""

        def execute(
            self,
            run_id: int,
            actual_parameters: Optional[Dict[str, Any]] = None,
        ) -> Optional[ExperimentRun]:
            """APPROVED -> EXECUTED.
            Records execution conditions.
            Does NOT run the experiment.
            """

        def complete(
            self,
            run_id: int,
            observation_refs: Optional[List[int]] = None,
        ) -> Optional[ExperimentRun]:
            """EXECUTED -> COMPLETED.
            Links observations.
            Does NOT evaluate.
            """

        def fail(
            self,
            run_id: int,
            reason: str,
        ) -> Optional[ExperimentRun]:
            """EXECUTED -> FAILED."""

        def cancel(
            self,
            run_id: int,
            reason: str,
        ) -> Optional[ExperimentRun]:
            """PROPOSED|APPROVED -> CANCELLED."""

        # QUERY
        def get(self, run_id: int) -> Optional[ExperimentRun]
        def list_by_status(self, status: str, limit: int = 100)
        def list_by_blueprint(self, tool_blueprint_id: str, limit: int = 100)
