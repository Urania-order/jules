    """
    Blockage Analysis Service — PHENOMENON-CENTRIC blockage analysis.

    Given a phenomenon_id, this service identifies constraints that
    may block the phenomenon, using existing DomainRelation records
    (TASK 07) and existing ontology models (TASK 02-04).

    IMPORTANT:
    - This service is READ-ONLY (no writes to relations or models).
    - Output is ANALYSIS, NOT prescription. No intervention is
      recommended. Human review is required for any action.
    - Complementary to ConstraintPerspective (TASK 08), which is
      constraint-centric. This service is phenomenon-centric.
    """

    from dataclasses import dataclass, field, asdict
    from typing import Any, Dict, List, Optional, Set, Tuple
    from sqlalchemy.orm import Session

    from smos.models.models import RelationType, EpistemicStatus
    from smos.models.constraint import Constraint
    from smos.models.context import Context
    from smos.services.domain_relation_service import DomainRelationService


    @dataclass
    class BlockageAnalysis:
        """Blockage analysis — ANALYSIS, NOT prescription."""
        kind: str = "blockage_analysis"
        phenomenon_id: Optional[int] = None
        direct_constraints: List[Dict[str, Any]] = field(default_factory=list)
        indirect_constraints: List[Dict[str, Any]] = field(default_factory=list)
        dependency_constraints: List[Dict[str, Any]] = field(default_factory=list)
        context_constraints: List[Dict[str, Any]] = field(default_factory=list)
        unknowns: List[Dict[str, Any]] = field(default_factory=list)
        epistemic_status: str = EpistemicStatus.HYPOTHESIZED.value
        note: str = (
            "Analysis only — NOT prescription. "
            "No intervention recommended. Human review required."
        )

        def to_dict(self) -> Dict[str, Any]:
            return asdict(self)


    class BlockageAnalysisService:
        BLOCKING_RELATION_TYPES = (
            RelationType.BLOCKS,
            RelationType.PREVENTS,
            RelationType.SUPPRESSES,
        )
        DEPENDENCY_RELATION_TYPES = (
            RelationType.REQUIRES,
            RelationType.DEPENDS_ON,
        )
        LOW_CONFIDENCE_THRESHOLD = 0.5

        def __init__(self, db: Session,
                     domain_service: Optional[DomainRelationService] = None):
            self.db = db
            self.domain_service = domain_service or DomainRelationService(db)

        # ---- Public API ----
        def analyze_blockage(
            self,
            phenomenon_id: int,
            indirect_max_hops: int = 2,
        ) -> BlockageAnalysis:
            """Analyze blockage for a phenomenon. READ-ONLY."""
            ...

        # ---- Helpers (testable, read-only) ----
        def _direct_blocking_constraints(self, phenomenon_id):
            """Constraints with BLOCKS/PREVENTS/SUPPRESSES -> phenomenon."""
            ...

        def _indirect_blocking_constraints(self, phenomenon_id, max_hops=2):
            """Constraints blocking intermediates that block the phenomenon.
            Multi-hop paths (bounded by max_hops). Documented depth limit.
            """
            ...

        def _dependency_constraints(self, phenomenon_id):
            """Phenomenon -> REQUIRES/DEPENDS_ON -> X; X blocked by constraints."""
            ...

        def _context_constraints(self, phenomenon_id):
            """Contexts hosting relevant constraints, or explicitly linked
            via DomainRelation (context <-> phenomenon, constraint <-> context).
            Also: match constraint.context (String) -> Context.name.
            """
            ...

        def _unknowns(self, direct, indirect, dependency):
            """Constraints with low confidence OR missing evidence."""
            ...

        def _constraint_dict(self, constraint_id, **extra):
            """Read-only dict of constraint for the output."""
            ...

DETECTION LOGIC (minimal, read-only)
============================================================

analyze_blockage(phenomenon_id, indirect_max_hops=2):

1. Direct:
   rels = domain_service.list_into("phenomenon", phenomenon_id)
   filter rels.relation_type in BLOCKING_RELATION_TYPES
   filter rels.source_type == "constraint"
   -> direct_constraints: [
        {"constraint_id": rel.source_id,
         "relation_type": rel.relation_type.value,
         "confidence": rel.confidence,
         "evidence": rel.evidence,
         "constraint_name": <fetched from Constraint.name>,
         "constraint_type": <fetched from Constraint.type.value>,
         "strength": <fetched from Constraint.strength>}
      ]

2. Indirect (bounded):
   For each intermediate target X (type="phenomenon") that is BLOCKED-by
   the phenomenon's blockers? — actually:
     Step A: get constraints C that BLOCK phenomenon (direct).
     Step B: for each C, find X such that C BLOCKS X, X BLOCKS phenomenon.
     Implementation: build a BFS up to max_hops over BLOCKING relations,
     starting from the phenomenon as target.
   -> indirect_constraints: [
        {"constraint_id": c_id,
         "via": {"type": "phenomenon", "id": via_id},
         "path": ["BLOCKS", "BLOCKS"],
         "hops": 2,
         "confidence": <min of rels' confidence>}
      ]
   Deduplicate by (constraint_id, via_id).

3. Dependency:
   deps = domain_service.list_for("phenomenon", phenomenon_id)
   filter deps.relation_type in DEPENDENCY_RELATION_TYPES
   For each dep target X (type="phenomenon" or "context"):
       blockers_of_X = domain_service.list_into(X.type, X.id)
       filter rels.relation_type in BLOCKING_RELATION_TYPES
       -> dependency_constraints: [
            {"constraint_id": c_id,
             "depends_via": {"type": X.type, "id": X.id},
             "relation_type": dep.relation_type.value,
             "confidence": <min>}
          ]

4. Context:
   - For each constraint in direct / indirect / dependency:
       * Look up Constraint by id, read .context (String)
       * Match Context.name == constraint.context
       * Also: DomainRelation where source_type="constraint", source_id=c_id,
         target_type="context"
   - Also: DomainRelation where source_type="context" and target=(phenomenon)
     with relation_type=BLOCKS
   -> context_constraints: [
        {"context_id": ctx.id,
         "context_name": ctx.name,
         "reason": "hosts_constraint" | "direct_block",
         "constraint_ids": [c_id, ...]}
      ]

5. Unknowns:
   From direct / indirect / dependency, collect constraints where:
     - confidence < LOW_CONFIDENCE_THRESHOLD
     - OR evidence is empty list
   -> unknowns: [
        {"constraint_id": c_id,
         "reason": "low_confidence" | "missing_evidence",
         "confidence": c.confidence,
         "threshold": LOW_CONFIDENCE_THRESHOLD}
      ]

6. Build BlockageAnalysis:
   kind = "blockage_analysis"
   epistemic_status = HYPOTHESIZED (always)
   note = "Analysis only — NOT prescription. ..."
   NO suggested_action / recommended_fix / priority_score.
