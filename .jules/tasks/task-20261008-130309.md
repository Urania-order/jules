    def semantic_neighbors(
        self,
        recipe_id: int,
        k: int = 10,
        entity_types: Optional[List[str]] = None,
    ) -> List[Dict[str, Any]]:
        """Return semantic neighbors of a recipe.

        CROSS-POLLINATION signal only. NOT evidence.
        NOT a mutation. NOT a DomainRelation.
        Read-only.
        """
        if entity_types is None:
            entity_types = self.DEFAULT_CANDIDATE_ENTITY_TYPES
        return self.semantic_service.similar(
            "recipe", recipe_id, k=k, entity_types=entity_types
        )

PART D — RecipeService.semantic_search (SIGNAL)

    def semantic_search(
        self,
        query: str,
        k: int = 10,
        entity_types: Optional[List[str]] = None,
    ) -> List[Dict[str, Any]]:
        """Semantic search for cross-pollination candidates. SIGNAL only."""
        if entity_types is None:
            entity_types = self.DEFAULT_CANDIDATE_ENTITY_TYPES
