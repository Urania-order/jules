1. Add shared JS helpers in frontend/index.html:

   function sortTasksByCreatedAt(tasks, desc = true) {
     const key = (t) => t.created_at || '';
     return [...tasks].sort((a, b) => {
       const ka = key(a), kb = key(b);
       if (!ka && !kb) return 0;
       if (!ka) return 1;
       if (!kb) return -1;
       return desc ? kb.localeCompare(ka) : ka.localeCompare(kb);
     });
   }

   function filterTasksByAuthor(tasks, author) {
     if (!author || author === 'all') return tasks;
     return tasks.filter(t => (t.proposed_by || 'unknown') === author);
   }

   function searchTasks(tasks, q) {
     if (!q) return tasks;
     const needle = q.toLowerCase();
     return tasks.filter(t => {
       const hay = [t.id, t.title, t.request, t.proposed_by]
         .filter(Boolean).join(' ').toLowerCase();
       return hay.includes(needle);
     });
   }

   function applyTaskFilters(tasks) {
     const author = document.getElementById('task-filter-author')?.value || 'all';
     const q = document.getElementById('task-filter-search')?.value || '';
     const order = document.getElementById('task-sort-order')?.value || 'desc';
     let out = tasks;
     out = filterTasksByAuthor(out, author);
     out = searchTasks(out, q);
     out = sortTasksByCreatedAt(out, order === 'desc');
     return out;
   }
